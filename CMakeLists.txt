# The following five lines of boilerplate have to be in your project's
# CMakeLists in this exact order for cmake to work correctly
cmake_minimum_required(VERSION 3.16)

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

# Set partition table
set(PARTITION_CSV_PATH ${CMAKE_CURRENT_SOURCE_DIR}/partitions.csv)

project(moon-phase)

# Automatically include SPIFFS data from spiffs/ folder
spiffs_create_partition_image(storage spiffs FLASH_IN_PROJECT)

# Moon phase assets: flash the equirect albedo blob into the moon_assets
# data partition (subtype "undefined" = plain raw binary region; read at
# boot via esp_partition_read). The blob is
# generated by the asset pipeline (local-only, gitignored):
#   cd assets/images/moon && python -m moon.assets
# It emits albedo.bin (MOON_ALB_W x MOON_ALB_H u8).
set(MOON_ALBEDO_BIN ${CMAKE_CURRENT_SOURCE_DIR}/assets/images/moon/albedo.bin)
if(EXISTS ${MOON_ALBEDO_BIN})
    esptool_py_flash_to_partition(flash "moon_assets" ${MOON_ALBEDO_BIN})
else()
    message(WARNING "moon albedo asset not found at ${MOON_ALBEDO_BIN} -- "
        "run 'cd assets/images/moon && python -m moon.assets' to generate it "
        "(the build succeeds, but the moon_assets partition is not flashed).")
endif()
